cache_root <- Sys.getenv("LOCALAPPDATA", unset = tools::R_user_dir("data4401-project2", "cache"))
data_dir <- file.path(cache_root, "data4401-project2", "home-credit-default-risk")

# Download a table only when it is not already available in the cache.
load_table <- function(name, ...) {
  if (!name %in% names(drive_file_ids)) {
    stop("Unknown table: ", name)
  }

  csv_path <- file.path(data_dir, paste0(name, ".csv"))

  if (!file.exists(csv_path)) {
    dir.create(data_dir, recursive = TRUE, showWarnings = FALSE)
    googledrive::drive_deauth()
    googledrive::drive_download(
      file = googledrive::as_id(drive_file_ids[[name]]),
      path = csv_path,
      overwrite = FALSE
    )
  }

  readr::read_csv(csv_path, show_col_types = FALSE, ...)
}

3. Logistic Regression Baseline

This baseline uses the stratified training and validation sets created above. The code uses tidyverse for data preparation and base R’s built-in glm() for logistic regression. All medians, modes, means, standard deviations, and category levels are learned from the training set only. The test set remains untouched for the final evaluation.

# Identify numeric and categorical columns from the training set.
numeric_features <- X_train_strat |>
  select(where(is.numeric)) |>
  names()

categorical_features <- X_train_strat |>
  select(where(Negate(is.numeric))) |>
  names()

# Store the training-set values needed for preprocessing.
numeric_medians <- X_train_strat |>
  summarise(across(all_of(numeric_features), ~ median(.x, na.rm = TRUE)))

# Return the category that appears most often in a variable.
most_frequent <- function(x) {
  observed <- x[!is.na(x)]
  names(sort(table(observed), decreasing = TRUE))[1]
}

categorical_modes <- X_train_strat |>
  summarise(across(all_of(categorical_features), most_frequent))

# Apply the same missing-value rules to training and validation data.
impute_data <- function(data) {
  data |>
    mutate(
      across(
        all_of(numeric_features),
        ~ replace_na(.x, numeric_medians[[cur_column()]])
      ),
      across(
        all_of(categorical_features),
        ~ replace_na(.x, categorical_modes[[cur_column()]])
      )
    )
}

train_processed <- impute_data(X_train_strat)
val_processed <- impute_data(X_val_strat)

# Treat validation categories not seen during training as the training mode.
for (column in categorical_features) {
  training_values <- unique(train_processed[[column]])
  val_processed[[column]][!val_processed[[column]] %in% training_values] <-
    categorical_modes[[column]]

  train_processed[[column]] <- factor(train_processed[[column]], levels = training_values)
  val_processed[[column]] <- factor(val_processed[[column]], levels = training_values)
}

# Standardize numeric columns using training-set means and standard deviations.
numeric_means <- train_processed |>
  summarise(across(all_of(numeric_features), mean))

numeric_sds <- train_processed |>
  summarise(across(all_of(numeric_features), sd)) |>
  mutate(across(everything(), ~ if_else(is.na(.x) | .x == 0, 1, .x)))

standardize_data <- function(data) {
# Subtract the training mean and divide by the training standard deviation.
  data |>
    mutate(
      across(
        all_of(numeric_features),
        ~ (.x - numeric_means[[cur_column()]]) / numeric_sds[[cur_column()]]
      )
    )
}

train_processed <- standardize_data(train_processed)
val_processed <- standardize_data(val_processed)

# model.matrix converts categorical columns to 0/1 dummy variables.
# Its default treatment coding leaves out one reference level per category.
matrix_terms <- terms(~ ., data = train_processed)
training_levels <- lapply(train_processed[categorical_features], levels)

X_train_matrix <- model.matrix(
  matrix_terms,
  data = train_processed,
  xlev = training_levels
)

X_val_matrix <- model.matrix(
  matrix_terms,
  data = val_processed,
  xlev = training_levels
)

# Fit an ordinary logistic regression without weights, tuning, or resampling.
baseline_fit <- glm.fit(
  x = X_train_matrix,
  y = y_train_strat,
  family = binomial()
)

# A redundant column can have an undefined coefficient; it contributes zero.
baseline_coefficients <- replace_na(baseline_fit$coefficients, 0)

validation_probability <- as.vector(
  plogis(X_val_matrix %*% baseline_coefficients)
)

# Use 0.50 as the cutoff for converting probabilities into class predictions.
validation_prediction <- if_else(validation_probability >= 0.50, 1, 0)

# Calculate the requested validation metrics.
true_positive <- sum(validation_prediction == 1 & y_val_strat == 1)
true_negative <- sum(validation_prediction == 0 & y_val_strat == 0)
false_positive <- sum(validation_prediction == 1 & y_val_strat == 0)
false_negative <- sum(validation_prediction == 0 & y_val_strat == 1)

precision_value <- if_else(
  true_positive + false_positive == 0,
  0,
  true_positive / (true_positive + false_positive)
)

recall_value <- true_positive / (true_positive + false_negative)
f1_value <- if_else(
  precision_value + recall_value == 0,
  0,
  2 * precision_value * recall_value / (precision_value + recall_value)
)

# The rank formula calculates ROC-AUC from predicted probabilities.
positive_ranks <- sum(rank(validation_probability)[y_val_strat == 1])
n_positive <- sum(y_val_strat == 1)
n_negative <- sum(y_val_strat == 0)
roc_auc_value <- (
  positive_ranks - n_positive * (n_positive + 1) / 2
) / (n_positive * n_negative)

validation_metrics <- tibble(
  metric = c("ROC-AUC", "Accuracy", "F1 score", "Precision", "Recall"),
  estimate = c(
    roc_auc_value,
    mean(validation_prediction == y_val_strat),
    f1_value,
    precision_value,
    recall_value
  )
)

validation_confusion_matrix <- table(
  Actual = factor(y_val_strat, levels = c(0, 1)),
  Predicted = factor(validation_prediction, levels = c(0, 1))
)
